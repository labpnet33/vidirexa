const audio=import("https://vireel-games-production.up.railway.app/packages/fold-groove/0.1.4/f04d2a37dcc6ea1b83f025f8b6de23e6cadb89dfa22df964abf21a353f64acf1/audio.mjs");
self.onmessage=async e=>{try{const {synthesize}=await audio,pcm=await synthesize(e.data);if(!pcm)self.postMessage({done:true,pcm:null});else self.postMessage({done:true,pcm:{left:pcm.left,right:pcm.right,sampleRate:pcm.sampleRate}},[pcm.left.buffer,pcm.right.buffer]);}
catch(error){self.postMessage({done:true,failed:String(error&&error.message||error)});}};
audio.then(()=>self.postMessage({ready:true}),error=>self.postMessage({done:true,failed:String(error&&error.message||error)}));
